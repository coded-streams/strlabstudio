SDK & Integration
Connect Str:::Beam to your application using the REST API or one of the SDK clients.
pom.xml
<dependency> <groupId>io.codedstreams.strbeam</groupId> <artifactId>strbeam-spring-boot-starter</artifactId> <version>1.0.0</version> </dependency>
application.yml
strbeam: engine-url: YOUR_ENGINE_URL tenant-key: YOUR_TENANT_KEY api-key: ${STRBEAM_API_KEY} catalog: YOUR_CATALOG
from strbeam import StrBeamClient client = StrBeamClient( engine_url = "YOUR_ENGINE_URL", tenant_key = "YOUR_TENANT_KEY", api_key = os.environ["STRBEAM_API_KEY"], ) session = client.open_session() op = client.submit(session.session_id, statement="SHOW TABLES;")
curl -X POST YOUR_ENGINE_URL/api/v1/sessions \ -H "Authorization: Bearer $STRBEAM_API_KEY" \ -H "Content-Type: application/json" -d '{}'
curl -X PUT YOUR_ENGINE_URL/api/v1/sessions/{sessionId}/heartbeat \ -H "Authorization: Bearer $STRBEAM_API_KEY"
STRBEAM_ENGINE_URL: http://strbeam:8090 STRBEAM_TENANT_KEY: YOUR_TENANT_KEY STRBEAM_API_KEY: ${STRBEAM_API_KEY}
